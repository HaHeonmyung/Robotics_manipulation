# 공통 평가: 프로젝트 루트에서 실행합니다.
# 48x8m 연속 혼합 험지, L10 난도 범위(0.9,1.0), 원본 Ant 보상,
# seed24 / 환경100 / 16초 / 높이 종료 ON.
# 높이 판정 = 몸통 Z - 출발 타일 origin Z < 0.31m (②와 같은 정의).
# 행동/모터/물리는 ②의 원본 Ant 설정을 모든 모델에 동일하게 적용합니다.
# tilt/flight/stuck/목표 종료는 없습니다. 코스 경계 이탈은 모든 모델에 적용합니다.
# 발 접촉 센서는 전진/체공 진단용이며 모든 모델의 정책 입력에서 제외합니다.
# 모델 입력을 자동 확인해 60 / 123(좁은맵) / 247(넓은맵) 환경을 선택합니다.
# 같은 지형 메시 및 초기 로봇 상태인지 해시로 확인하고, 다르면 비교를 중단합니다.

conda activate lerobot-arena
cd ~/IsaacLab

# 현재 남아 있는 5개 모델을 한 번에 평가합니다.
# ④는 커리큘럼+넓은 RayCaster 모델(model_3300.pt)입니다.
# 체크포인트는 ①~③ 999 / ④ 3300 / ⑤ 2800이며, 훈련 지형·보상·예산이 다릅니다.
# 이 비교는 개발 과정의 결과이며, 각 기법의 효과만을 분리한 누적 실험은 아닙니다.
./isaaclab.sh -p scripts/reinforcement_learning/rsl_rl/evaluate_ant_common.py \
  --headless --seed 24 --num_envs 100 --duration 16 \
  --checkpoint logs/rsl_rl/ant/2026-09-17_13-20-13_ant_baseline/model_999.pt --label 1_flat \
  --checkpoint logs/rsl_rl/ant_terrain_hard/2026-10-05_12-24-39_hard_seed42/model_999.pt --label 2_hard \
  --checkpoint logs/rsl_rl/ant_terrain_curriculum/2026-10-06_19-08-42_curriculum_only_seed42/model_999.pt --label 3_curriculum \
  --checkpoint logs/rsl_rl/ant_continuous_wide_ray/2026-10-05_21-25-33_wide_ray_fresh_seed42_10k/model_3300.pt --label 4_curriculum_ray \
  --checkpoint logs/rsl_rl/ant_continuous_wide_ray_recovery/2026-10-06_14-05-48_recovery_seed42_10k/model_2800.pt --label 5_recovery \
  --output_dir logs/ant_terrain_comparison/common_16s_height_on_seed24

# 단일 모델 평가: checkpoint만 바꾸면 센서 구성은 자동 선택됩니다.
./isaaclab.sh -p scripts/reinforcement_learning/rsl_rl/evaluate_ant_common.py \
  --headless --seed 24 --num_envs 100 --duration 16 \
  --checkpoint logs/rsl_rl/ant_terrain_curriculum/2026-10-06_19-08-42_curriculum_only_seed42/model_999.pt \
  --label 3_curriculum

# 높이 종료 OFF 비교: 첫 명령에 --disable_torso_height를 추가하고
# --output_dir를 logs/ant_terrain_comparison/common_16s_height_off_seed24로 변경합니다.
# 60초 비교: --duration 60으로 바꾸면 192x8m / 24개 타일 코스를 사용합니다.
# 16초/60초, 높이ON/OFF 결과는 서로 다른 조건이므로 별도로 비교합니다.

# GUI: 센서 없는 ①②③은 Isaac-Ant-Common-Eval-v0,
# 좁은 RayCaster 모델은 Isaac-Ant-Common-Ray-Eval-v0,
# 넓은 RayCaster 모델은 Isaac-Ant-Common-Wide-Ray-Eval-v0를 사용합니다.
./isaaclab.sh -p scripts/reinforcement_learning/rsl_rl/play.py \
  --task Isaac-Ant-Common-Wide-Ray-Eval-v0 \
  --num_envs 8 --seed 24 --camera_mode overview --log_resets --real-time \
  --checkpoint logs/rsl_rl/ant_continuous_wide_ray_recovery/2026-10-06_14-05-48_recovery_seed42_10k/model_2800.pt

# GUI 높이 종료 OFF는 명령 끝에 env.terminations.torso_height=null 추가.
